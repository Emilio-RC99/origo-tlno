cmake_minimum_required(VERSION 3.16)
project(OrigoTLNO LANGUAGES CXX)

set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

include(CTest)
find_package(OpenMP REQUIRED COMPONENTS CXX)

set(TLNO_SIMD "baseline" CACHE STRING "Instruction set: baseline, avx2, avx512, or native")
set_property(CACHE TLNO_SIMD PROPERTY STRINGS baseline avx2 avx512 native)
set(TLNO_TUNE "generic" CACHE STRING "CPU tuning for explicit x86 ISA builds")
set(TLNO_LTO_JOBS "" CACHE STRING "GCC LTO worker count; empty uses CMake's IPO defaults")
set(TLNO_NUCLEAR_LANES "" CACHE STRING "Optional nuclear lanes: 1, 2, 4 or 8")
option(TLNO_BUILD_ONEE_TOOLS "Build one-electron validation and benchmark tools" ON)
option(TLNO_BUILD_TWOE_TOOLS "Build two-electron benchmark tools" ON)
option(TLNO_ENABLE_PYTHON_VALIDATION "Register available NumPy/PySCF/mpmath validation checks" ON)
option(TLNO_FAST_MATH "Compile with -ffast-math" OFF)
option(TLNO_ENABLE_LTO "Enable interprocedural optimization" OFF)

add_library(tlno_build_options INTERFACE)
target_compile_features(tlno_build_options INTERFACE cxx_std_17)
target_include_directories(tlno_build_options INTERFACE "${PROJECT_SOURCE_DIR}")
target_link_libraries(tlno_build_options INTERFACE OpenMP::OpenMP_CXX)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    string(TOLOWER "${CMAKE_SYSTEM_PROCESSOR}" tlno_processor)
    if(NOT tlno_processor MATCHES "^(x86_64|amd64|x64)$" AND NOT TLNO_SIMD STREQUAL "baseline")
        message(FATAL_ERROR "Explicit x86 ISA selection requires an x86-64 host; use baseline for scalar builds.")
    endif()
    if(TLNO_SIMD STREQUAL "baseline")
        if(tlno_processor MATCHES "^(x86_64|amd64|x64)$")
            target_compile_options(tlno_build_options INTERFACE -march=x86-64 -mtune=${TLNO_TUNE})
        endif()
    elseif(TLNO_SIMD STREQUAL "avx2")
        target_compile_options(tlno_build_options INTERFACE -march=x86-64-v3 -mtune=${TLNO_TUNE})
    elseif(TLNO_SIMD STREQUAL "avx512")
        target_compile_options(tlno_build_options INTERFACE -march=x86-64-v4 -mtune=${TLNO_TUNE})
    elseif(TLNO_SIMD STREQUAL "native")
        target_compile_options(tlno_build_options INTERFACE -march=native)
    else()
        message(FATAL_ERROR "TLNO_SIMD must be baseline, avx2, avx512, or native")
    endif()
    if(TLNO_FAST_MATH)
        target_compile_options(tlno_build_options INTERFACE -ffast-math)
        target_link_options(tlno_build_options INTERFACE -ffast-math)
    endif()
elseif(NOT TLNO_SIMD STREQUAL "baseline" OR TLNO_FAST_MATH)
    message(FATAL_ERROR "Explicit ISA and fast-math options require GCC or Clang.")
endif()
if(NOT TLNO_NUCLEAR_LANES STREQUAL "")
    if(NOT TLNO_NUCLEAR_LANES MATCHES "^(1|2|4|8)$")
        message(FATAL_ERROR "TLNO_NUCLEAR_LANES must be empty, 1, 2, 4, or 8")
    endif()
    target_compile_definitions(tlno_build_options INTERFACE TLNO_NUCLEAR_LANES=${TLNO_NUCLEAR_LANES})
endif()
if(TLNO_ENABLE_LTO)
    include(CheckIPOSupported)
    check_ipo_supported(RESULT tlno_ipo_supported OUTPUT tlno_ipo_error)
    if(NOT tlno_ipo_supported)
        message(FATAL_ERROR "LTO is unavailable: ${tlno_ipo_error}")
    endif()
    if(NOT TLNO_LTO_JOBS STREQUAL "")
        if(NOT CMAKE_CXX_COMPILER_ID STREQUAL "GNU" OR NOT TLNO_LTO_JOBS MATCHES "^[1-9][0-9]*$")
            message(FATAL_ERROR "TLNO_LTO_JOBS requires GCC and a positive worker count")
        endif()
        target_compile_options(tlno_build_options INTERFACE -flto=${TLNO_LTO_JOBS})
        target_link_options(tlno_build_options INTERFACE -flto=${TLNO_LTO_JOBS})
    else()
        set(CMAKE_INTERPROCEDURAL_OPTIMIZATION TRUE)
    endif()
endif()

set(CMAKE_POSITION_INDEPENDENT_CODE ON)
# One output folder, where the Python scripts look for the binaries by default.
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")

add_subdirectory(core)
add_subdirectory(integrals/one_electron)
add_subdirectory(integrals/two_electron)

if(TLNO_BUILD_ONEE_TOOLS OR TLNO_BUILD_TWOE_TOOLS)
    add_subdirectory(benchmarks/common)
endif()

if(TLNO_BUILD_ONEE_TOOLS)
    add_subdirectory(benchmarks/one_electron_engine)
endif()

if(TLNO_BUILD_TWOE_TOOLS)
    add_subdirectory(benchmarks/two_electron_engine)
endif()
